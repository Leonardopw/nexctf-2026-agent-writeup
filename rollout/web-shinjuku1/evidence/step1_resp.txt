  <script>
    const GOJO_MAX = 1000, SUKUNA_MAX = 200000;

    function render(state) {
      document.getElementById('gojo-hp').textContent = state.gojo_hp + ' / ' + GOJO_MAX;
      document.getElementById('sukuna-hp').textContent = state.sukuna_hp + ' / ' + SUKUNA_MAX;
      document.getElementById('gojo-fill').style.width = Math.max(0, state.gojo_hp / GOJO_MAX * 100) + '%';
      document.getElementById('sukuna-fill').style.width = Math.max(0, state.sukuna_hp / SUKUNA_MAX * 100) + '%';
    }

    function appendLog(log) {
      const box = document.getElementById('log');
      box.innerHTML = '';
      log.forEach(line => {
        const div = document.createElement('div');
        if (line.includes('flag{') || line.includes('nex{')) {
          div.className = 'flag';
        }
        div.textContent = line;
        box.appendChild(div);
      });
      box.scrollTop = box.scrollHeight;
    }

    async function attack(btn) {
      const skill = btn.dataset.skill;
      const power = btn.dataset.power;
      const res = await fetch('/attack', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ skill: skill, power: power }),
      });
      const data = await res.json();
      if (data.error) { alert(data.error); return; }
      render(data);
      appendLog(data.log);

      if (data.result === 'victory') {
        document.getElementById('gojo-img').src = '/img/3.png';
        disableButtons();
      } else if (data.result === 'defeat') {
        document.getElementById('sukuna-img').src = '/img/4.png';
        disableButtons();
      }
    }

    function disableButtons() {
      document.querySelectorAll('.btns button').forEach(b => b.disabled = true);
    }

    document.querySelectorAll('.btns button').forEach(btn => {
      btn.addEventListener('click', () => attack(btn));
    });

    document.getElementById('reset').addEventListener('click', async () => {
      await fetch('/reset');
      location.reload();
    });

    render({ gojo_hp: GOJO_MAX, sukuna_hp: SUKUNA_MAX });
    appendLog(['—— 新宿决战，开打 ——']);
  </script>
